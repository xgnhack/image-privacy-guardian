# 资源优化实施方案

## 1. 实施概述

基于当前代码分析，软件在处理大量图片时存在以下关键问题：
- `monitoring_manager.py` 中固定的 `max_concurrent_threads = 3`
- `main.py` 中的 `initial_scan` 方法缺乏分批处理
- 缺乏内存监控和自动清理机制
- UI线程可能被长时间阻塞

## 2. 具体实施步骤

### 2.1 第一阶段：监控管理器优化

#### 2.1.1 动态线程池管理

**文件：** `monitoring_manager.py`

**修改位置：** `MonitoringManager.__init__()` 方法

**新增方法：**
```python
def adjust_thread_pool_size(self):
    """动态调整线程池大小"""
    current_time = time.time()
    if current_time - self.last_adjustment_time < self.adjustment_interval:
        return
        
    try:
        # 获取系统资源使用情况
        cpu_percent = psutil.cpu_percent(interval=1)
        memory_percent = psutil.virtual_memory().percent
        
        # 调整策略
        if cpu_percent > 80 or memory_percent > 85:
            # 高负载时减少线程
            new_max = max(self.min_threads, self.current_max_threads - 1)
        elif cpu_percent < 50 and memory_percent < 60 and len(self.processing_threads) >= self.current_max_threads:
            # 低负载且队列满时增加线程
            new_max = min(self.max_threads, self.current_max_threads + 1)
        else:
            new_max = self.current_max_threads
            
        if new_max != self.current_max_threads:
            self.current_max_threads = new_max
            self.log_message.emit(f"🔧 线程池大小调整为: {new_max}")
            
        self.last_adjustment_time = current_time
        
    except Exception as e:
        self.log_message.emit(f"⚠️ 线程池调整失败: {str(e)}")
```

#### 2.1.2 队列管理优化

**修改 `queue_file_for_processing` 方法：**
```python
def queue_file_for_processing(self, file_path: str, event_type: str, priority: str = 'normal'):
    """将文件加入处理队列（支持优先级）"""
    if not self.is_running:
        return
        
    try:
        # 动态调整线程池
        self.adjust_thread_pool_size()
        
        # 清理已完成的线程
        self.cleanup_finished_threads()
        
        # 检查并发限制（使用动态值）
        if len(self.processing_threads) >= self.current_max_threads:
            # 根据优先级决定处理策略
            if priority == 'high':
                self.log_message.emit(f"🚨 高优先级任务等待处理: {os.path.basename(file_path)}")
                # 可以考虑中断低优先级任务
            else:
                self.log_message.emit(f"⏳ 处理队列已满，跳过: {os.path.basename(file_path)}")
                return
                
        # 创建处理工作器
        worker = ProcessingWorker(file_path, event_type, self)
        
        # 在新线程中执行
        thread = threading.Thread(target=worker.process_file, daemon=True)
        thread.start()
        
        self.processing_threads.append(thread)
        self.increment_processed()
        
    except Exception as e:
        self.log_message.emit(f"❌ 提交处理任务失败: {str(e)}")
```

#### 2.1.3 内存监控集成

**新增内存管理类：**
```python
class MemoryMonitor:
    def __init__(self, monitoring_manager):
        self.monitoring_manager = monitoring_manager
        self.memory_limit_mb = 1024  # 1GB限制
        self.warning_threshold = 0.8
        self.critical_threshold = 0.9
        self.last_cleanup_time = 0
        self.cleanup_interval = 60  # 60秒检查一次
        
    def check_memory_usage(self):
        """检查内存使用情况"""
        try:
            process = psutil.Process()
            memory_mb = process.memory_info().rss / 1024 / 1024
            usage_ratio = memory_mb / self.memory_limit_mb
            
            current_time = time.time()
            
            if usage_ratio > self.critical_threshold:
                self.emergency_cleanup()
                self.monitoring_manager.log_message.emit(
                    f"🚨 内存使用过高 ({memory_mb:.1f}MB), 执行紧急清理"
                )
            elif usage_ratio > self.warning_threshold and current_time - self.last_cleanup_time > self.cleanup_interval:
                self.gentle_cleanup()
                self.monitoring_manager.log_message.emit(
                    f"⚠️ 内存使用警告 ({memory_mb:.1f}MB), 执行清理"
                )
                self.last_cleanup_time = current_time
                
            return usage_ratio
            
        except Exception as e:
            self.monitoring_manager.log_message.emit(f"❌ 内存监控失败: {str(e)}")
            return 0
            
    def gentle_cleanup(self):
        """温和的内存清理"""
        import gc
        gc.collect()
        
    def emergency_cleanup(self):
        """紧急内存清理"""
        import gc
        gc.collect()
        # 可以考虑暂停新任务处理
        # 清理可能的缓存
```

### 2.2 第二阶段：主程序扫描优化

#### 2.2.1 分批扫描实现

**文件：** `main.py`

**修改 `ScanWorker` 类：**
```python
class ScanWorker(QThread):
    # ... 现有信号定义 ...
    batch_progress = pyqtSignal(dict)  # 新增批次进度信号
    
    def __init__(self, folders, manager):
        super().__init__()
        self.folders = folders
        self.manager = manager
        self.should_stop = False
        
        # 分批处理配置
        self.batch_size = 20  # 每批处理20个文件
        self.batch_delay = 200  # 批次间延迟200ms
        self.memory_check_interval = 10  # 每10个文件检查一次内存
        
    def scan_folder_optimized(self, folder_path):
        """优化的文件夹扫描方法"""
        if not os.path.exists(folder_path):
            return 0
            
        try:
            # 首先收集所有图像文件
            all_image_files = []
            for root, dirs, files in os.walk(folder_path):
                if self.should_stop:
                    break
                    
                for file in files:
                    if self.should_stop:
                        break
                        
                    file_path = os.path.join(root, file)
                    if self.manager.is_image_file(file_path):
                        all_image_files.append(file_path)
            
            total_files = len(all_image_files)
            if total_files == 0:
                return 0
                
            self.progress_update.emit(f"📊 发现 {total_files} 个图像文件，开始分批处理...")
            
            # 分批处理
            processed_count = 0
            for i in range(0, total_files, self.batch_size):
                if self.should_stop:
                    break
                    
                batch = all_image_files[i:i + self.batch_size]
                batch_result = self.process_batch(batch, i + 1, total_files)
                processed_count += batch_result
                
                # 批次间休息
                if i + self.batch_size < total_files:  # 不是最后一批
                    self.msleep(self.batch_delay)
                    
                # 内存检查
                if hasattr(self.manager, 'memory_monitor'):
                    memory_usage = self.manager.memory_monitor.check_memory_usage()
                    if memory_usage > 0.8:  # 内存使用超过80%
                        self.progress_update.emit("⏳ 内存使用过高，延长休息时间...")
                        self.msleep(self.batch_delay * 2)
                        
            return processed_count
            
        except Exception as e:
            self.error_occurred.emit(f"扫描文件夹失败 {folder_path}: {str(e)}")
            return 0
    
    def process_batch(self, file_batch, batch_start, total_files):
        """处理单个批次"""
        processed_in_batch = 0
        
        for idx, file_path in enumerate(file_batch):
            if self.should_stop:
                break
                
            current_index = batch_start + idx - 1
            progress_percent = int((current_index / total_files) * 100)
            
            self.progress_update.emit(
                f"📄 处理文件 ({current_index}/{total_files}, {progress_percent}%): {os.path.basename(file_path)}"
            )
            
            # 检查是否已处理过
            if not self.manager.is_file_processed(file_path):
                self.file_found.emit(f"🆕 发现未处理文件: {os.path.basename(file_path)}")
                
                # 使用队列处理而不是直接处理
                self.manager.queue_file_for_processing(file_path, "首次扫描", priority="low")
                processed_in_batch += 1
                
            # 文件间小延迟
            self.msleep(50)
            
        # 发送批次完成信号
        self.batch_progress.emit({
            'batch_processed': len(file_batch),
            'batch_found': processed_in_batch,
            'total_progress': int((batch_start + len(file_batch) - 1) / total_files * 100)
        })
        
        return processed_in_batch
```

#### 2.3.2 用户控制功能

**添加暂停/恢复功能：**
```python
class ProcessingController:
    def __init__(self, monitoring_manager):
        self.monitoring_manager = monitoring_manager
        self.is_paused = False
        self.pause_lock = threading.Lock()
        
    def pause_processing(self):
        """暂停处理"""
        with self.pause_lock:
            self.is_paused = True
            self.monitoring_manager.log_message.emit("⏸️ 处理已暂停")
            
    def resume_processing(self):
        """恢复处理"""
        with self.pause_lock:
            self.is_paused = False
            self.monitoring_manager.log_message.emit("▶️ 处理已恢复")
            
    def wait_if_paused(self):
        """如果暂停则等待"""
        while self.is_paused:
            time.sleep(0.1)
```
